\begin{abstract}\noindent
\textbf{The Reggae Shark Protocol (v1.2.0)} is a verifiable execution kernel and multi-cloud AI gateway that binds agent prompt dispatches to four mathematical and cryptographic verification layers: (1) an append-only, SHA-256 hash-chained and RFC~8032 Ed25519-signed \textbf{IntentSeal Verifiable Research Kernel (VRK v0.3.1)} ledger enforcing the \texttt{intentseal-vrk-claim-v2} schema, prompt/response SHA-256 commitments, on-disk evidence anchor checks, and an $N$-way commutative, associative, and idempotent Merkle-DAG join-semilattice for distributed split-brain reconciliation; (1B) a thread-synchronized \textbf{Request-Lifecycle Typestate and Worker-Epoch Isolation} state machine (\texttt{TypestateLifecycle.lean}, 12 zero-\texttt{sorry} theorems) that rules out partial-initialization payload dereferencing and cross-epoch worker-slot reads across multi-step event traces within the modelled state machine; (2) a bounded-LRU per-tenant \textbf{4D Soft-Wall ODE Cognitive Dynamics Governor} $(\gamma, H_{\text{obs}}, \dot{I}, \Psi)$ backed by 17 Lean 4 4D boundary theorems (\texttt{ExistenceVerification.lean}), 9 3D soft-wall theorems (\texttt{BoundaryVerification.lean}), and a subnormal-safe Patankar--Euler interval integrator; and (3) a \textbf{Bandlimited Low-Rank Spectral Context Memory (LRSC)} operator ($M=99, K=11$) with dual compressed/uncompressed SHA-256 sealing and a bounded-domain local attention sensitivity bound.
\end{abstract}

\vspace{-6pt}\section{Layer 2 \& Layer 3 Mathematical Equations}\vspace{-3pt}
The Layer 2 governor evolves $x(t) = (\gamma(t), H_{\text{obs}}(t), \dot{I}(t), \Psi(t))$ under the smooth ($C^\infty$) soft-wall system \cite{existence2026}:
{\small\begin{align}
  \frac{d\gamma}{dt} &= a \dot{I} \mathcal{E}_{\text{avail}}\left(1 - \frac{\gamma}{\gamma_{\max}}\right) - b\gamma, \label{eq:gamma}\\
  \frac{dH_{\text{obs}}}{dt} &= c\gamma\left(H_{\max} - H_{\text{obs}}\right) - \dot{I}_{\text{erase}}\tanh\!\left(\frac{H_{\text{obs}}}{\epsilon_H}\right), \label{eq:hobs}\\
  \frac{d\dot{I}}{dt} &= e\tanh\!\left(\frac{\dot{I}}{\epsilon_I}\right)\frac{dH_{\text{obs}}}{dt} + f\mathcal{P}_{\text{eff}}(\gamma, \Psi) - g\dot{I}, \label{eq:idot}\\
  \frac{d\Psi}{dt} &= \kappa_{\text{meta}} R_{\text{self}}(x)\left(1 - \Psi\right) - \mu_\Psi \Psi, \label{eq:psi}
\end{align}}
where $\mathcal{P}_{\text{eff}}(\gamma, \Psi) = \mathcal{P}_0 + \lambda_{\text{sim}} r_{\text{LRSC}} \gamma + \lambda_{\text{lucid}} \Psi$ and $R_{\text{self}}(x) = \text{fb}(x)^2 / (\text{fb}(x)^2 + (f\mathcal{P}_0)^2 + \sigma_0^2)$ with $\text{fb}(x) = e\tanh(\dot{I}/\epsilon_I)(dH_{\text{obs}}/dt)$.

\vspace{-6pt}\section{Five Core Verification Guards \& Five Edge-Case Probes (30/30 Tests)}\vspace{-3pt}
\begin{enumerate}\setlength{\itemsep}{2pt}\setlength{\parskip}{0pt}
  \item \textbf{Guard 1 \& Probe A (Patankar--Euler Box Invariance \& Python Subnormal Floor):} \texttt{step\_ode\_patankar\_interval} updates each coordinate via the cleared-denominator scheme
  \begin{equation}
    x_{k+1} = \frac{x_k + \Delta t\, P(x_k)\, x_{\max}}{1 + \Delta t\,(P(x_k) + D(x_k))}, \qquad P(x_k) \ge 0,\; D(x_k) \ge 0,\; 1 + \Delta t(P+D) \ge 1, \label{eq:patankar}
  \end{equation}
  paired with $10^{-12}$ Taylor cutoffs and a Python runtime float64 floor \texttt{FLOAT64\_NORMAL\_MIN} $= 2.225\times 10^{-308}$ whenever $x_k > 0$ or $P(x_k) > 0$. Lean 4 Theorems \texttt{patankar\_euler\_step\_in\_box} and \texttt{patankar\_strict\_positivity\_floor} prove the real-arithmetic bounds over $\mathbb{R}$ (\texttt{test\_15}, \texttt{test\_21}).
  \item \textbf{Guard 2 \& Probe B (Pointwise Decay Bound, Bounded Tenant LRU \& Waking Drive Floor):} Lean 4 Theorem \texttt{psi\_decay\_floor\_bound} proves the pointwise derivative inequality $d\Psi/dt \ge -\mu_\Psi \Psi$, which by the classical Grönwall lemma yields $\Psi(t+\Delta t) \ge \Psi(t)e^{-\mu_\Psi \Delta t}$. \texttt{MultiProviderGateway} bounds per-tenant engines in an \texttt{OrderedDict} LRU cache (\texttt{max\_tenants=128}) and enforces $\mathcal{P}_0 \ge 1.0 \gg 10/99$ (\texttt{test\_16}, \texttt{test\_22}, \texttt{test\_29}).
  \item \textbf{Guard 3 \& Probe C (Prompt-Committed Bandlimited LRSC Witness \& Local Sensitivity Proxy):} \texttt{LRSCCompressor.compress} injects a baseline anchor on slot 0 to eliminate scalar-multiple normalization collisions (distinguishing single-byte and constant-byte strings in \texttt{test\_27}), projects onto $V_{\text{ring}}$ ($\sigma_{\min}(S_{11}) = 0.18164624 > 0$ \cite{lrsc2026}), evaluates the local bounded-domain sensitivity proxy $L_{\text{local}}(L_W, d_k)\eta_\perp(\tilde{x})$ on $\|\tilde{x}\|_2 \le 1$ \cite{kim2021} (Theorem \texttt{lrsc\_attention\_lipschitz\_error\_bound}), and seals \texttt{prompt\_sha256}, \texttt{response\_sha256}, \texttt{lrsc\_sha256}, and \texttt{uncompressed\_ring\_sha256} (\texttt{test\_17}, \texttt{test\_23}, \texttt{test\_27}).
  \item \textbf{Guard 4 \& Probe D (Inductive Typestate Trace Safety, Absorbing \texttt{ZEROED\_FREED} \& RLock Order):} Lean 4 Theorems 1--3 and 5 prove by induction over arbitrary event lists \texttt{evs : List LifecycleEvent} that \texttt{ZeroedFreed} is strictly absorbing and that \texttt{\{AbortedPoisoned, ZeroedFreed\}} never reaches \texttt{SealedActive} or \texttt{ShutdownClean}, while Python enforces \texttt{WorkerEpochGuard.\_lock $\to$ RequestSlot.\_lock} (\texttt{test\_18}, \texttt{test\_24}, \texttt{test\_30}).
  \item \textbf{Guard 5 \& Probe E (Ed25519-Signed Ledger, Multi-Parent Verification \& Associative Merkle-DAG):} Every claim is signed with RFC~8032 Ed25519 (\texttt{test\_28}), \texttt{verify\_chain\_list} rejects forged multi-parent entries not chained from \texttt{VRK\_TRUST\_ROOT\_DIGEST} (\texttt{test\_26}), and \texttt{merge\_divergent\_ledgers} satisfies $(A\sqcup B)\sqcup C = A\sqcup(B\sqcup C)$ and $A\sqcup A = A$ (Theorem \texttt{merkle\_dag\_semilattice\_laws}, \texttt{test\_19}, \texttt{test\_20}, \texttt{test\_25}).
\end{enumerate}
