\begin{abstract}\noindent
\textbf{The Reggae Shark Protocol (v1.3.0)} is a verifiable execution kernel and multi-cloud AI gateway that binds agent prompt dispatches to four verification layers and three cross-program mathematical/security upgrades: (1) an append-only, SHA-256 hash-chained and RFC~8032 Ed25519-signed \textbf{IntentSeal Verifiable Research Kernel (VRK v0.3.1)} ledger enforcing \texttt{intentseal-vrk-claim-v2}, prompt/response commitments, on-disk evidence anchor checks, an $N$-way associative Merkle-DAG join-semilattice, and \textbf{Track B multi-principal OS subprocess key isolation} (\texttt{IsolatedSignerSubprocess}); (1B) a thread-synchronized \textbf{Request-Lifecycle Typestate and Worker-Epoch Isolation} state machine (\texttt{TypestateLifecycle.lean}, expanded to 15 zero-\texttt{sorry} theorems); (2) a bounded-LRU per-tenant \textbf{4D Soft-Wall ODE Governor} $(\gamma, H_{\text{obs}}, \dot{I}, \Psi)$ backed by 17 4D boundary theorems (\texttt{ExistenceVerification.lean}), 9 3D soft-wall theorems (\texttt{BoundaryVerification.lean}), and \textbf{Track A 200-bit outward-rounded \texttt{mpmath.iv} Patankar--Euler interval enclosures} plus \textbf{Track C 120-segment piecewise Duhamel contraction certificates} for Navier--Stokes \texttt{WP19}; and (3) a \textbf{Bandlimited LRSC} operator ($M=99, K=11$) with a bounded-domain local sensitivity bound.
\end{abstract}

\vspace{-7pt}\section{Layer 2 Equations, Five Core Guards \& Three Cross-Program Upgrades (34/34 Tests)}\vspace{-3pt}
The Layer 2 governor evolves $x(t) = (\gamma(t), H_{\text{obs}}(t), \dot{I}(t), \Psi(t))$ under the smooth ($C^\infty$) soft-wall system \cite{existence2026}:
{\small\begin{align}
  \frac{d\gamma}{dt} &= a \dot{I} \mathcal{E}_{\text{avail}}\left(1 - \frac{\gamma}{\gamma_{\max}}\right) - b\gamma, \qquad
  \frac{dH_{\text{obs}}}{dt} = c\gamma\left(H_{\max} - H_{\text{obs}}\right) - \dot{I}_{\text{erase}}\tanh\!\left(\frac{H_{\text{obs}}}{\epsilon_H}\right), \label{eq:gamma_hobs}\\
  \frac{d\dot{I}}{dt} &= e\tanh\!\left(\frac{\dot{I}}{\epsilon_I}\right)\frac{dH_{\text{obs}}}{dt} + f\mathcal{P}_{\text{eff}}(\gamma, \Psi) - g\dot{I}, \qquad
  \frac{d\Psi}{dt} = \kappa_{\text{meta}} R_{\text{self}}(x)\left(1 - \Psi\right) - \mu_\Psi \Psi, \label{eq:idot_psi}
\end{align}}
where $\mathcal{P}_{\text{eff}}(\gamma, \Psi) = \mathcal{P}_0 + \lambda_{\text{sim}} r_{\text{LRSC}} \gamma + \lambda_{\text{lucid}} \Psi$, $R_{\text{self}}(x) = \text{fb}(x)^2 / (\text{fb}(x)^2 + (f\mathcal{P}_0)^2 + \sigma_0^2)$, and $\text{fb}(x) = e\tanh(\dot{I}/\epsilon_I)(dH_{\text{obs}}/dt)$.
\begin{enumerate}\setlength{\itemsep}{1.2pt}\setlength{\parskip}{0pt}
  \item \textbf{Guard 1, Probe A \& Track A (200-Bit \texttt{mpmath.iv} Patankar--Euler Real-to-Float Bridge):} Updates each coordinate via
  \begin{equation}
    x_{k+1} = \frac{x_k + \Delta t\, P(x_k)\, x_{\max}}{1 + \Delta t\,(P(x_k) + D(x_k))}, \qquad P(x_k) \ge 0,\; D(x_k) \ge 0,\; 1 + \Delta t(P+D) \ge 1, \label{eq:patankar}
  \end{equation}
  paired with Python float64 floor \texttt{FLOAT64\_NORMAL\_MIN} $= 2.225\times 10^{-308}$ and \texttt{step\_\allowbreak ode\_\allowbreak patankar\_\allowbreak mp\_\allowbreak interval} (200-bit outward-rounded \texttt{mpmath.iv} intervals of width $< 10^{-45}$ agreeing with float64 within $< 10^{-12}$, Theorem 13 \texttt{patankar\_\allowbreak interval\_\allowbreak rounding\_\allowbreak in\_\allowbreak box}, \texttt{test\_15}, \texttt{test\_21}, \texttt{test\_31}).
  \item \textbf{Guard 2 \& Probe B (Pointwise Decay Bound, Bounded Tenant LRU \& Waking Drive Floor):} Theorem \texttt{psi\_decay\_floor\_bound} proves $d\Psi/dt \ge -\mu_\Psi \Psi$ (yielding $\Psi(t+\Delta t) \ge \Psi(t)e^{-\mu_\Psi \Delta t}$ via Grönwall), while \texttt{MultiProviderGateway} enforces \texttt{max\_tenants=128} and $\mathcal{P}_0 \ge 1.0 \gg 10/99$ (\texttt{test\_16}, \texttt{test\_22}, \texttt{test\_29}).
  \item \textbf{Guard 3 \& Probe C (Prompt-Committed Bandlimited LRSC Witness \& Local Sensitivity Proxy):} \texttt{LRSCCompressor.compress} injects a slot-0 anchor to eliminate scalar-multiple collisions (\texttt{test\_27}), projects onto $V_{\text{ring}}$ ($\sigma_{\min}(S_{11}) = 0.18164624 > 0$ \cite{lrsc2026}), bounds $L_{\text{local}}\eta_\perp(\tilde{x})$ on $\|\tilde{x}\|_2 \le 1$ \cite{kim2021} (Theorem 10), and seals \texttt{prompt\_sha256} and \texttt{response\_sha256} (\texttt{test\_17}, \texttt{test\_23}, \texttt{test\_27}).
  \item \textbf{Guard 4 \& Probe D (Inductive Typestate Trace Safety, Absorbing \texttt{ZEROED\_FREED} \& RLock Order):} Theorems 1--5 prove over arbitrary event lists \texttt{evs : List LifecycleEvent} that \texttt{ZeroedFreed} is strictly absorbing and \texttt{\{AbortedPoisoned, ZeroedFreed\}} never reaches \texttt{SealedActive} or \texttt{ShutdownClean}, with Python \texttt{guard.\_lock $\to$ slot.\_lock} (\texttt{test\_18}, \texttt{test\_24}, \texttt{test\_30}).
  \item \textbf{Guard 5, Probe E \& Track B (Ed25519 Signing, Multi-Principal Subprocess Isolation \& Merkle-DAG):} Every claim is signed with Ed25519 (\texttt{test\_28}), \texttt{IsolatedSignerSubprocess} isolates the private key in a child OS process across a narrow IPC pipe (Theorem 14 \texttt{principal\_\allowbreak key\_\allowbreak isolation\_\allowbreak noninterference}, \texttt{test\_32}), \texttt{verify\_chain\_list} rejects forged multi-parent tails (\texttt{test\_26}), and \texttt{merge\_divergent\_ledgers} satisfies $(A\sqcup B)\sqcup C = A\sqcup(B\sqcup C)$ (Theorem 12, \texttt{test\_19}, \texttt{test\_25}).
  \item \textbf{Track C (120-Segment Piecewise Duhamel Contraction Bridge for Navier--Stokes \texttt{WP19}):} Theorem 15 (\texttt{duhamel\_\allowbreak piecewise\_\allowbreak contraction\_\allowbreak bound}) and \texttt{certify\_\allowbreak wp19\_\allowbreak piecewise\_\allowbreak duhamel\_\allowbreak bridge} prove in 200-bit \texttt{mpmath.iv} arithmetic across the 120 segments (\href{https://doi.org/10.5281/zenodo.23034545}{10.5281/zenodo.23034545}) that piecewise dissipative propagation $E_{k+1} = e^{\mu_{\text{eff}}\Delta t}E_k + \delta_k$ suppresses the $1.6364\times 10^{14}$ whole-path Grönwall remainder below the signed $C_{500}$ lower bound $5.7586\times 10^9$ whenever $\mu_{\text{eff}} < \mu_{\text{crit}} \approx 22.95\,\text{s}^{-1}$ (\texttt{test\_33}, \texttt{test\_34}).
