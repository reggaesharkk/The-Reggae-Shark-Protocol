\begin{abstract}
\noindent
\textbf{The Reggae Shark Protocol (v1.3.2)} is a verifiable execution kernel and multi-cloud AI gateway that binds agent prompt dispatches to four verification layers: (1) an append-only, SHA-256 hash-chained and RFC 8032 Ed25519-signed \textbf{IntentSeal Verifiable Research Kernel (VRK v0.3.1)} ledger requiring an explicit private seed or child-read key file, mandatory pinned \texttt{expected\_pubkey} verification, append-only \texttt{fsync} persistence, timestamped signed head commitments (\texttt{intentseal-vrk-head-v2} with \texttt{issued\_at}) and minimum-length export verification (\texttt{verify\_exported\_ledger}), an $N$-way associative Merkle-DAG join-semilattice, and stateful child-process key isolation (\texttt{IsolatedSignerSubprocess} enforcing \texttt{prev\_hash == last\_signed\_digest}); (1B) a thread-synchronized \textbf{Request-Lifecycle Typestate and Worker-Epoch Isolation} state machine (\texttt{TypestateLifecycle.lean}, 16 zero-\texttt{sorry} theorems including unique reachability of \texttt{SealedActive} from \texttt{Uninit}) verified by differential trace testing and CI \texttt{\#eval} diffing against \texttt{data/lean\_transition\_table.json}; (2) a per-tenant \textbf{4D Soft-Wall ODE Governor} $(\gamma, H_{\text{obs}}, \dot{I}, \Psi)$ backed by 17 4D boundary theorems (\texttt{ExistenceVerification.lean}), 9 3D soft-wall theorems (\texttt{BoundaryVerification.lean}), runtime \texttt{float64} box checks (\texttt{inside\_B4\_checked}), and \textbf{200-bit outward-rounded \texttt{mpmath.iv} Patankar--Euler interval enclosures} (\texttt{inside\_B4\_mp\_iv\_certified}), plus an illustrative 120-segment Duhamel parameter study on \texttt{data/wp19\_segments.json}; and (3) a \textbf{Bandlimited LRSC} operator ($M=99, K=11$) with linear perturbation budget scaling (\texttt{perturbation\_budget\_scaling}).
\end{abstract}

\section{What Is and Is Not Modelled (Epistemic Claim Firewall \& Threat Model)}
\vspace{-0.2em}

\colorbox{boxbg}{\parbox{0.985\textwidth}{\footnotesize
\textbf{\textcolor{navy}{Epistemic Standard (Build. Attack. Falsify. Preserve.):}} Every theorem and runtime guard in this repository is bounded by explicit domain assumptions, \texttt{SECURITY.md} threat boundaries, and \texttt{scope.does\_not\_imply} declarations.}}

\begin{itemize}
  \item \textbf{Application-Layer Typestate \& Stateful Subprocess Signer vs.\ OS Ring-0:} \texttt{RequestSlot} and \texttt{WorkerEpochGuard} enforce typestate invariants in Python, while \texttt{IsolatedSignerSubprocess} loads its key from a child-read \texttt{seed\_file} (or generates an ephemeral key in-child) and refuses to sign any entry with \texttt{prev\_hash $\neq$ last\_signed\_digest}. If the parent gateway is compromised mid-session, it can still request signatures on new forward entries extending \texttt{last\_signed\_digest}, but cannot rewrite history, fork an old digest, sign a truncated head, or extract the raw key. Theorem 14 (\texttt{projection\_independent\_of\_secret}) is a definitional projection lemma; process isolation rests on the runtime child-process design (\texttt{test\_32}, \texttt{test\_41}).
  \item \textbf{Real Arithmetic in Lean vs.\ 200-Bit \texttt{mpmath.iv} \& IEEE-754 Runtime Checks:} Lean 4 Theorems 6, 7, 9, 10, 13, and 15 are proved over $\mathbb{R}$; \texttt{step\_ode\_patankar\_interval} performs a fast \texttt{float64} 2-ULP check (\texttt{inside\_B4\_checked}), whereas \texttt{step\_ode\_patankar\_mp\_interval} bridges discrete steps to $\mathbb{R}$ via 200-bit outward-rounded \texttt{mpmath.iv} intervals (\texttt{inside\_B4\_mp\_iv\_certified}, \texttt{INTERVAL\_CERTIFIED\_RESULT}).
  \item \textbf{Illustrative WP19 Parameter Study vs.\ Continuum Navier--Stokes:} \texttt{evaluate\_wp19\_piecewise\_duhamel\_study} evaluates the 120-segment Duhamel recurrence on the assumed segment rates loaded from \texttt{data/wp19\_segments.json} (SHA-256: \texttt{a78a67c8...99319}, \href{https://doi.org/10.5281/zenodo.23034545}{10.5281/zenodo.23034545}, \texttt{NUMERICAL\_OBSERVATION}); in-repo hash pinning catches accidental file corruption, while external tamper resistance is anchored by the release notes and Zenodo record. Continuum 3D Navier--Stokes regularity remains \texttt{OPEN}.
\end{itemize}

\section{Layer 2 Equations \& Verification Suite (42/42 Tests Passing)}
\vspace{-0.2em}

The Layer 2 governor evolves $x(t) = (\gamma(t), H_{\text{obs}}(t), \dot{I}(t), \Psi(t))$ under the smooth ($C^\infty$) soft-wall system \cite{upadhyay2026existence}:
\begin{align}
\frac{d\gamma}{dt} &= a \dot{I} \mathcal{E}_{\text{avail}}\left(1 - \frac{\gamma}{\gamma_{\max}}\right) - b\gamma, &
\frac{dH_{\text{obs}}}{dt} &= c\gamma\left(H_{\max} - H_{\text{obs}}\right) - \dot{I}_{\text{erase}}\tanh\!\left(\frac{H_{\text{obs}}}{\epsilon_H}\right), \label{eq:gh}\\
\frac{d\dot{I}}{dt} &= e\tanh\!\left(\frac{\dot{I}}{\epsilon_I}\right)\frac{dH_{\text{obs}}}{dt} + f\mathcal{P}_{\text{eff}}(\gamma,\Psi) - g\dot{I}, &
\frac{d\Psi}{dt} &= \kappa_{\text{meta}} R_{\text{self}}(x)\left(1 - \Psi\right) - \mu_\Psi \Psi, \label{eq:ipsi}
\end{align}
where $\mathcal{P}_{\text{eff}}(\gamma,\Psi) = \mathcal{P}_0 + \lambda_{\text{sim}} r_{\text{LRSC}}\gamma + \lambda_{\text{lucid}}\Psi$, $R_{\text{self}}(x) = \text{fb}(x)^2 / (\text{fb}(x)^2 + (f\mathcal{P}_0)^2 + \sigma_0^2)$, and $\text{fb}(x) = e\tanh(\dot{I}/\epsilon_I)(dH_{\text{obs}}/dt)$.

\begin{enumerate}
  \item \textbf{Patankar--Euler Real-to-Float Bridge (200-Bit \texttt{mpmath.iv} Certification vs.\ \texttt{float64} Check):} Updates each coordinate via
  \begin{equation}
  x_{k+1} = \frac{x_k + \Delta t\,P(x_k)\,x_{\max}}{1 + \Delta t\,(P(x_k) + D(x_k))}, \qquad P(x_k) \ge 0,\; D(x_k) \ge 0,\; 1 + \Delta t(P+D) \ge 1,
  \end{equation}
  paired with Python \texttt{float64} floor \texttt{FLOAT64\_NORMAL\_MIN} $= 2.225 \times 10^{-308}$ (\texttt{inside\_B4\_checked}) and \texttt{step\_ode\_patankar\_mp\_interval} (200-bit outward-rounded \texttt{mpmath.iv} intervals of width $< 10^{-45}$ agreeing with \texttt{float64} within $< 10^{-12}$, \texttt{inside\_B4\_mp\_iv\_certified}, Theorem 13 \texttt{patankar\_interval\_rounding\_in\_box}, \texttt{test\_15}, \texttt{test\_21}, \texttt{test\_31}).
  \item \textbf{Pointwise Decay Bound, Token Registry Authentication, Per-Tenant Persistence \& Non-Evicting Cap:} Theorem \texttt{psi\_decay\_floor\_bound} proves $d\Psi/dt \ge -\mu_\Psi\Psi$ (yielding $\Psi(t+\Delta t) \ge \Psi(t)e^{-\mu_\Psi\Delta t}$ via Gr\"onwall), while \texttt{MultiProviderGateway} authenticates Bearer tokens via \texttt{sha256(token) $\to$ tenant\_id} (HTTP 401 on unknown token), persists per-tenant ledgers to \texttt{<ledger\_dir>/\allowbreak{}<tenant\_id>.jsonl}, returns HTTP 503 at \texttt{max\_tenants=128} without eviction, rate-limits requests (HTTP 429), and clamps $\mathcal{P}_0 \ge 1.0 \gg 10/99$ (\texttt{test\_16}, \texttt{test\_22}, \texttt{test\_29}, \texttt{test\_38}, \texttt{test\_40}).
  \item \textbf{Prompt-Committed Bandlimited LRSC Witness \& Perturbation Budget Scaling:} \texttt{LRSCCompressor.\allowbreak{}compress} injects a slot-0 anchor to prevent scalar-multiple collisions (\texttt{test\_27}), projects onto $V_{\text{ring}}$ ($\sigma_{\min}(S_{11}) = 0.18164624 > 0$ \cite{upadhyay2026lrsc}), evaluates linear perturbation budget scaling $L_{\text{local}}\eta_\perp(\tilde{x})$ on $\|\tilde{x}\|_2 \le 1$ \cite{kim2021lipschitz} (Theorem 10 \texttt{perturbation\_budget\_scaling}), and is verified against empirical softmax attention perturbations (\texttt{test\_17}, \texttt{test\_23}, \texttt{test\_27}, \texttt{test\_38}).
  \item \textbf{Inductive Typestate Trace Safety, Unique Reachability \& Differential Equivalence:} Theorems 1--5 and 16 prove over arbitrary event lists \texttt{evs : List LifecycleEvent} that \texttt{ZeroedFreed} is strictly absorbing, $\{\texttt{AbortedPoisoned}, \texttt{ZeroedFreed}\}$ is closed, and \texttt{runTrace Uninit evs = SealedActive} $\leftrightarrow$ \texttt{[BeginInit,\allowbreak{} SealSuccess]} (\texttt{sealed\_active\_unique\_trace\_from\_uninit}), while \texttt{test\_36} and \texttt{test\_42} verify equivalence against \texttt{data/lean\_transition\_table.json}.
  \item \textbf{Pinned-Key Ed25519 Signing, Timestamped Head Anti-Replay, Stateful Child Signer \& Merkle-DAG:} \texttt{verify\_exported\_ledger} requires a timestamped \texttt{signed\_head} (\texttt{intentseal-vrk-head-v2} with \texttt{issued\_at}) and \texttt{min\_length} (reference head hash in \texttt{data/release\_head.json}: \texttt{866e1919b5a3e9e1...6608ee5b}, \texttt{test\_35}, \texttt{test\_39}), \texttt{IsolatedSignerSubprocess} loads its key from \texttt{seed\_file} and enforces \texttt{prev\_hash == last\_signed\_digest} (\texttt{test\_32}, \texttt{test\_41}), and \texttt{merge\_divergent\_ledgers} satisfies $(A \sqcup B) \sqcup C = A \sqcup (B \sqcup C)$ (Theorem 12, \texttt{test\_19}, \texttt{test\_25}).
  \item \textbf{Illustrative 120-Segment Piecewise Duhamel Parameter Study (\texttt{data/wp19\_segments.json}):} Theorem 15 (\texttt{duhamel\_piecewise\_contraction\_bound}) proves $\alpha(\alpha e_0 + d_1) + d_2 < G(G e_0 + d_1) + d_2$ for $0 \le \alpha < G$ and $d_1 > 0$, and \texttt{evaluate\_wp19\_piecewise\_duhamel\_study} evaluates the 120-segment recurrence in 200-bit \texttt{mpmath.iv} arithmetic (\href{https://doi.org/10.5281/zenodo.23034545}{10.5281/zenodo.23034545}, \texttt{test\_33}, \texttt{test\_34}).
\end{enumerate}
