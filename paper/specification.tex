\section{Layer 1 \& 1B: VRK Hash-Chaining and Formal \texttt{php2shell} Typestate Immunity}

\begin{theorem}[Typestate OOM Poisoning \& Epoch Isolation --- \texttt{TypestateLifecycle.lean}]

  $\delta(\mathsf{PartialInit}, \mathsf{OOMAbort}) = \mathsf{AbortedPoisoned}$.
  \item \textbf{\texttt{CVE-2026-109618} Payload \& Destructor UAF Knockout (\texttt{no\_deref\_after\_partial\_oom\_abort}, \texttt{no\_shutdown\_uaf\_on\_poisoned\_state}):}
  Neither $\mathsf{canDereferencePayload}$ nor $\mathsf{canRunCleanDestructor}$ holds on $\delta(\mathsf{PartialInit}, \mathsf{OOMAbort})$.
  \item \textbf{\texttt{CVE-2026-109619} Stale Worker-Epoch Knockout (\texttt{stale\_epoch\_read\_rejected}):}
  For any worker at epoch $e_{\mathrm{worker}} \in \mathbb{N}$ and any slot $w$ with $w.\mathsf{slotEpoch} \neq e_{\mathrm{worker}}$, $\neg \mathsf{canAccessWorkerSlot}(e_{\mathrm{worker}}, w)$.
