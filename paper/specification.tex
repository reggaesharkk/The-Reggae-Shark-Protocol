\begin{abstract}
\noindent
\textbf{The Reggae Shark Protocol (v1.3.1)} is a verifiable execution kernel and multi-cloud AI gateway that binds agent prompt dispatches to four verification layers: (1) an append-only, SHA-256 hash-chained and RFC 8032 Ed25519-signed \textbf{IntentSeal Verifiable Research Kernel (VRK v0.3.1)} ledger requiring an explicit private seed (\texttt{SHARK\_ED25519\_SEED\_HEX}), mandatory pinned \texttt{expected\_pubkey} verification, append-only \texttt{fsync} persistence, signed head records (\texttt{intentseal-vrk-head-v1}) to detect tail truncation, an $N$-way associative Merkle-DAG join-semilattice, and optional child-process key isolation (\texttt{IsolatedSignerSubprocess}); (1B) a thread-synchronized \textbf{Request-Lifecycle Typestate and Worker-Epoch Isolation} state machine (\texttt{TypestateLifecycle.lean}, 15 zero-\texttt{sorry} theorems) verified by differential trace testing against its Lean \texttt{\#eval} JSON export; (2) a bounded-LRU per-tenant \textbf{4D Soft-Wall ODE Governor} $(\gamma, H_{\text{obs}}, \dot{I}, \Psi)$ backed by 17 4D boundary theorems (\texttt{ExistenceVerification.lean}), 9 3D soft-wall theorems (\texttt{BoundaryVerification.lean}), and \textbf{200-bit outward-rounded \texttt{mpmath.iv} Patankar--Euler interval enclosures}, plus an illustrative 120-segment Duhamel parameter study on \texttt{data/wp19\_segments.json}; and (3) a \textbf{Bandlimited LRSC} operator ($M=99, K=11$) with linear perturbation budget scaling (\texttt{perturbation\_budget\_scaling}).
\end{abstract}

\section{What Is and Is Not Modelled (Epistemic Claim Firewall \& Threat Model)}
\vspace{-0.2em}

\colorbox{boxbg}{\parbox{0.985\textwidth}{\footnotesize
\textbf{\textcolor{navy}{Epistemic Standard (Build. Attack. Falsify. Preserve.):}} Every theorem and runtime guard in this repository is bounded by explicit domain assumptions, \texttt{SECURITY.md} threat boundaries, and \texttt{scope.does\_not\_imply} declarations.}}

\begin{itemize}
  \item \textbf{Application-Layer Typestate \& Subprocess Signer vs.\ OS Ring-0:} \texttt{RequestSlot} and \texttt{WorkerEpochGuard} enforce typestate invariants in Python, while \texttt{IsolatedSignerSubprocess} isolates the Ed25519 private key in a child OS process (\texttt{subprocess.Popen}) across a pipe IPC boundary. They are \textit{not} an OS ring-0 kernel module or hardware HSM. Theorem 14 (\texttt{projection\_independent\_of\_secret}) is a definitional projection lemma; process isolation rests on the runtime child-process design and \texttt{test\_32}.
  \item \textbf{Real Arithmetic in Lean vs.\ 200-Bit \texttt{mpmath.iv} \& IEEE-754 Runtime Enclosures:} Lean 4 Theorems 6, 7, 9, 10, 13, and 15 are proved over $\mathbb{R}$; \texttt{step\_ode\_patankar\_mp\_interval} bridges discrete steps to $\mathbb{R}$ via 200-bit outward-rounded \texttt{mpmath.iv} intervals (\texttt{INTERVAL\_CERTIFIED\_RESULT}).
  \item \textbf{Illustrative WP19 Parameter Study vs.\ Continuum Navier--Stokes:} \texttt{evaluate\_wp19\_piecewise\_duhamel\_study} evaluates the 120-segment Duhamel recurrence on the assumed segment rates loaded from \texttt{data/wp19\_segments.json} (\href{https://doi.org/10.5281/zenodo.23034545}{10.5281/zenodo.23034545}, \texttt{NUMERICAL\_OBSERVATION}); it does not certify the Galerkin chain while segment rates are assumed, and continuum 3D Navier--Stokes regularity remains \texttt{OPEN}.
\end{itemize}

\section{Layer 2 Equations, Core Verification Guards \& Audit Remediations (38/38 Tests)}
\vspace{-0.2em}

The Layer 2 governor evolves $x(t) = (\gamma(t), H_{\text{obs}}(t), \dot{I}(t), \Psi(t))$ under the smooth ($C^\infty$) soft-wall system \cite{upadhyay2026existence}:
\begin{align}
\frac{d\gamma}{dt} &= a \dot{I} \mathcal{E}_{\text{avail}}\left(1 - \frac{\gamma}{\gamma_{\max}}\right) - b\gamma, &
\frac{dH_{\text{obs}}}{dt} &= c\gamma\left(H_{\max} - H_{\text{obs}}\right) - \dot{I}_{\text{erase}}\tanh\!\left(\frac{H_{\text{obs}}}{\epsilon_H}\right), \label{eq:gh}\\
\frac{d\dot{I}}{dt} &= e\tanh\!\left(\frac{\dot{I}}{\epsilon_I}\right)\frac{dH_{\text{obs}}}{dt} + f\mathcal{P}_{\text{eff}}(\gamma,\Psi) - g\dot{I}, &
\frac{d\Psi}{dt} &= \kappa_{\text{meta}} R_{\text{self}}(x)\left(1 - \Psi\right) - \mu_\Psi \Psi, \label{eq:ipsi}
\end{align}
where $\mathcal{P}_{\text{eff}}(\gamma,\Psi) = \mathcal{P}_0 + \lambda_{\text{sim}} r_{\text{LRSC}}\gamma + \lambda_{\text{lucid}}\Psi$, $R_{\text{self}}(x) = \text{fb}(x)^2 / (\text{fb}(x)^2 + (f\mathcal{P}_0)^2 + \sigma_0^2)$, and $\text{fb}(x) = e\tanh(\dot{I}/\epsilon_I)(dH_{\text{obs}}/dt)$.

\begin{enumerate}
  \item \textbf{Guard 1, Probe A \& Track A (200-Bit \texttt{mpmath.iv} Patankar--Euler Real-to-Float Bridge):} Updates each coordinate via
  \begin{equation}
  x_{k+1} = \frac{x_k + \Delta t\,P(x_k)\,x_{\max}}{1 + \Delta t\,(P(x_k) + D(x_k))}, \qquad P(x_k) \ge 0,\; D(x_k) \ge 0,\; 1 + \Delta t(P+D) \ge 1,
  \end{equation}
  paired with Python \texttt{float64} floor \texttt{FLOAT64\_NORMAL\_MIN} $= 2.225 \times 10^{-308}$ and \texttt{step\_ode\_patankar\_mp\_interval} (200-bit outward-rounded \texttt{mpmath.iv} intervals of width $< 10^{-45}$ agreeing with \texttt{float64} within $< 10^{-12}$, Theorem 13 \texttt{patankar\_interval\_rounding\_in\_box}, \texttt{test\_15}, \texttt{test\_21}, \texttt{test\_31}).
  \item \textbf{Guard 2 \& Probe B (Pointwise Decay Bound, Token-Bound Tenant LRU, Rate Limit \& Waking Drive Floor):} Theorem \texttt{psi\_decay\_floor\_bound} proves $d\Psi/dt \ge -\mu_\Psi\Psi$ (yielding $\Psi(t+\Delta t) \ge \Psi(t)e^{-\mu_\Psi\Delta t}$ via Gr\"onwall), while \texttt{MultiProviderGateway} binds HTTP tenants to Bearer tokens, enforces \texttt{max\_tenants=128}, rate-limits requests (HTTP 429), and clamps $\mathcal{P}_0 \ge 1.0 \gg 10/99$ (\texttt{test\_16}, \texttt{test\_22}, \texttt{test\_29}, \texttt{test\_38}).
  \item \textbf{Guard 3 \& Probe C (Prompt-Committed Bandlimited LRSC Witness \& Perturbation Budget Scaling):} \texttt{LRSCCompressor.compress} injects a slot-0 anchor to prevent scalar-multiple collisions (\texttt{test\_27}), projects onto $V_{\text{ring}}$ ($\sigma_{\min}(S_{11}) = 0.18164624 > 0$ \cite{upadhyay2026lrsc}), evaluates linear perturbation budget scaling $L_{\text{local}}\eta_\perp(\tilde{x})$ on $\|\tilde{x}\|_2 \le 1$ \cite{kim2021lipschitz} (Theorem 10 \texttt{perturbation\_budget\_scaling}), and is tested against empirical softmax attention perturbations on random weight matrices (\texttt{test\_17}, \texttt{test\_23}, \texttt{test\_27}, \texttt{test\_38}).
  \item \textbf{Guard 4, Probe D \& Task E (Inductive Typestate Trace Safety, Absorbing \texttt{ZEROED\_FREED} \& Differential Trace Equivalence):} Theorems 1--5 prove over arbitrary event lists \texttt{evs : List LifecycleEvent} that \texttt{ZeroedFreed} is strictly absorbing and $\{\texttt{AbortedPoisoned}, \texttt{ZeroedFreed}\}$ never reaches \texttt{SealedActive} or \texttt{ShutdownClean}, while \texttt{test\_36} verifies state-for-state equivalence between Python \texttt{RequestSlot} and \texttt{data/lean\_transition\_table.json} across 200 random traces (\texttt{test\_18}, \texttt{test\_24}, \texttt{test\_30}, \texttt{test\_36}).
  \item \textbf{Guard 5, Probe E \& Tasks A/B (Mandatory Pinned-Key Ed25519 Signing, Signed Head Record, Subprocess Isolation \& Merkle-DAG):} \texttt{Ed25519AuthoritySigner} requires \texttt{cryptography} and an explicit seed (rejecting unconfigured or retired \texttt{v1.2.0} seeds, \texttt{test\_28}, \texttt{test\_35}), \texttt{verify\_chain\_list} requires keyword-only \texttt{expected\_pubkey} and verifies \texttt{signed\_head} against tail truncation (\texttt{test\_26}, \texttt{test\_35}), \texttt{IsolatedSignerSubprocess} isolates or generates the signing key in a child OS process (\texttt{test\_32}), and \texttt{merge\_divergent\_ledgers} satisfies $(A \sqcup B) \sqcup C = A \sqcup (B \sqcup C)$ (Theorem 12, \texttt{test\_19}, \texttt{test\_25}).
  \item \textbf{Track C \& Task F (Illustrative 120-Segment Piecewise Duhamel Parameter Study on \texttt{data/wp19\_segments.json}):} Theorem 15 (\texttt{duhamel\_piecewise\_contraction\_bound}) proves $\alpha(\alpha e_0 + d_1) + d_2 < G(G e_0 + d_1) + d_2$ for $0 \le \alpha < G$ and $d_1 > 0$, and \texttt{evaluate\_wp19\_piecewise\_duhamel\_study} evaluates the 120-segment recurrence in 200-bit \texttt{mpmath.iv} arithmetic using the SHA-256-verified parameters in \texttt{data/wp19\_segments.json} (\href{https://doi.org/10.5281/zenodo.23034545}{10.5281/zenodo.23034545}, \texttt{test\_33}, \texttt{test\_34}).
\end{enumerate}
